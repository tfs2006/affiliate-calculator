            <canvas id="metricsChart"></canvas>

    <script>
    let chartInstance = null;

    async function calculate() {
        try {
            // Clear previous chart
            if (chartInstance) chartInstance.destroy();
            
            // Show loading
            document.getElementById('results').innerHTML = '<div style="padding: 1rem; text-align: center;">Calculating... ⏳</div>';
            
            // Get values
            const data = getInputValues();
            
            // Basic validation
            if (data.clicks <= 0 || data.conversions < 0) {
                throw new Error('❌ Invalid input: Clicks must be > 0\nConversions cannot be negative');
            }

            // Save to localStorage
            localStorage.setItem('lastCalc', JSON.stringify(data));

            // API call
            const response = await fetch('/calculate', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify(data)
            });

            if (!response.ok) throw new Error(`HTTP Error: ${response.status}`);
            
            const result = await response.json();
            
            if (result.status === 'error') throw new Error(result.message);
            
            displayResults(result.results);
            createChart(result.results);

        } catch (error) {
            document.getElementById('results').innerHTML = `
                <div style="color: var(--danger); padding: 1rem; background: #fee2e2; border-radius: 0.5rem;">
                    ${error.message}
                </div>
            `;
        }
    }

    function getInputValues() {
        return {
            clicks: Number(document.getElementById('clicks').value),
            conversions: Number(document.getElementById('conversions').value),
            investment: Number(document.getElementById('investment').value),
            revenue: Number(document.getElementById('revenue').value),
            expenses: Number(document.getElementById('expenses').value),
            acquisition_cost: Number(document.getElementById('acquisition_cost').value),
            ltv: Number(document.getElementById('ltv').value)
        };
    }

    function displayResults(results) {
        let html = `
            <div style="display: grid; grid-template-columns: repeat(2, 1fr); gap: 1rem;">
                <div style="padding: 1rem; background: #f0f9ff; border-radius: 0.5rem;">
                    <h3>📈 Conversion Metrics</h3>
                    <p>Conversion Rate: <strong>${results.conversion_rate?.toFixed(2) || 0}%</strong></p>
                    <p>Cost/Conversion: <strong>$${results.cost_per_conversion?.toFixed(2) || 0}</strong></p>
                </div>
                
                <div style="padding: 1rem; background: #f0fdf4; border-radius: 0.5rem;">
                    <h3>💰 Financial Metrics</h3>
                    <p>ROI: <strong>${results.roi?.toFixed(2) || 0}%</strong></p>
                    <p>Profit Margin: <strong>${results.profit_margin?.toFixed(2) || 0}%</strong></p>
                </div>
                
                <div style="padding: 1rem; background: #fff7ed; border-radius: 0.5rem;">
                    <h3>👥 Customer Metrics</h3>
                    <p>LTV:CAC Ratio: <strong>${results.ltv_cac_ratio?.toFixed(2) || 0}</strong></p>
                </div>
            </div>
        `;
        
        document.getElementById('results').innerHTML = html;
    }

    function createChart(results) {
        const ctx = document.getElementById('metricsChart').getContext('2d');
        
        chartInstance = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: ['Conversion Rate (%)', 'ROI (%)', 'Profit Margin (%)', 'LTV:CAC Ratio'],
                datasets: [{
                    label: 'Performance Metrics',
                    data: [
                        results.conversion_rate,
                        results.roi,
                        results.profit_margin,
                        results.ltv_cac_ratio
                    ],
                    backgroundColor: [
                        '#3b82f6',
                        '#10b981',
                        '#f59e0b',
                        '#ef4444'
                    ],
                    borderWidth: 0
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                scales: {
                    y: {
                        beginAtZero: true
                    }
                },
                plugins: {
                    legend: {
                        position: 'top',
                    },
                    title: {
                        display: true,
                        text: 'Performance Overview'
                    }
                }
            }
        });
    }

    function resetForm() {
        document.querySelectorAll('input').forEach(input => input.value = '');
        localStorage.removeItem('lastCalc');
        document.getElementById('results').innerHTML = '';
        if (chartInstance) chartInstance.destroy();
    }

    // Load previous session
    window.addEventListener('load', () => {
        const savedData = localStorage.getItem('lastCalc');
        if (savedData) {
            const data = JSON.parse(savedData);
            Object.entries(data).forEach(([key, value]) => {
                document.getElementById(key).value = value;
            });
        }
    });
    </script>
